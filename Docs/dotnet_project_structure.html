namespace ChangelogSaaS.Domain.Entities;

public class ChangelogEntry
{
    public Guid Id { get; private set; }
    public Guid ProjectId { get; private set; }
    public string Title { get; private set; } = "";
    public string ContentHtml { get; private set; } = "";
    public List<string> Tags { get; private set; } = [];
    public EntryStatus Status { get; private set; }
    public string? Version { get; private set; }
    public DateTime? PublishedAt { get; private set; }
    public DateTime CreatedAt { get; private set; }

    // Factory method — không dùng constructor public
    public static ChangelogEntry Create(
        Guid projectId, string title, string html,
        List<string> tags, string? version)
    {
        return new ChangelogEntry
        {
            Id = Guid.NewGuid(),
            ProjectId = projectId,
            Title = title,
            ContentHtml = html,
            Tags = tags,
            Version = version,
            Status = EntryStatus.Draft,
            CreatedAt = DateTime.UtcNow
        };
    }

    public void Publish()
    {
        if (Status == EntryStatus.Published)
            throw new DomainException("Entry already published");
        Status = EntryStatus.Published;
        PublishedAt = DateTime.UtcNow;
    }
}

PublishEntryCommand.cs
// MediatR CQRS — Command + Handler + Validator

public record PublishEntryCommand(Guid EntryId, Guid UserId)
    : IRequest;

public class PublishEntryHandler(
    IAppDbContext db,
    IEmailService email,
    ICacheService cache)
    : IRequestHandler<PublishEntryCommand>
{
    public async Task Handle(
        PublishEntryCommand cmd,
        CancellationToken ct)
    {
        var entry = await db.ChangelogEntries
            .Include(e => e.Project)
            .FirstOrDefaultAsync(e =>
                e.Id == cmd.EntryId &&
                e.Project.UserId == cmd.UserId, ct)
            ?? throw new NotFoundException("Entry not found");

        entry.Publish(); // Domain logic
        await db.SaveChangesAsync(ct);

        BackgroundJob.Enqueue<SendEmailNotificationJob>(
            j => j.Execute(entry.Id));

        await cache.RemoveAsync(
            $"widget:{entry.Project.Slug}");
    }
}

// CreateEntryCommand.cs
public record CreateEntryCommand(
    Guid ProjectId, Guid UserId,
    string Title, string ContentHtml,
    List<string> Tags, string? Version)
    : IRequest<Guid>;

public class CreateEntryHandler(IAppDbContext db)
    : IRequestHandler<CreateEntryCommand, Guid>
{
    public async Task<Guid> Handle(
        CreateEntryCommand cmd, CancellationToken ct)
    {
        var entry = ChangelogEntry.Create(
            cmd.ProjectId, cmd.Title,
            cmd.ContentHtml, cmd.Tags, cmd.Version);

        db.ChangelogEntries.Add(entry);
        await db.SaveChangesAsync(ct);
        return entry.Id;
    }
}

// CreateEntryCommandValidator.cs — chạy tự động qua ValidationBehavior
public class CreateEntryCommandValidator
    : AbstractValidator<CreateEntryCommand>
{
    public CreateEntryCommandValidator()
    {
        RuleFor(x => x.Title).NotEmpty().MaximumLength(200);
        RuleFor(x => x.ContentHtml).NotEmpty();
    }
}

StripeService.cs — Webhook
public class StripeService(AppDbContext db,
    IOptions<StripeSettings> opts)
    : IBillingService
{
    public async Task HandleWebhookAsync(
        string payload, string signature)
    {
        var evt = EventUtility.ConstructEvent(
            payload, signature,
            opts.Value.WebhookSecret);

        switch (evt.Type)
        {
            case "customer.subscription.created":
            case "customer.subscription.updated":
                await UpsertSubscriptionAsync(
                    evt.Data.Object as Subscription);
                break;

            case "customer.subscription.deleted":
                await CancelSubscriptionAsync(
                    evt.Data.Object as Subscription);
                break;
        }
    }

    public async Task<string> CreateCheckoutAsync(
        Guid userId, string priceId)
    {
        var options = new SessionCreateOptions
        {
            Mode = "subscription",
            LineItems = [new() {
                Price = priceId, Quantity = 1 }],
            SuccessUrl = "https://app.io/billing/success",
            CancelUrl = "https://app.io/billing",
            Metadata = new() {{ "userId", userId.ToString() }}
        };
        var session = await new SessionService()
            .CreateAsync(options);
        return session.Url;
    }
}

            Assembly.GetExecutingAssembly());
    }

// EmailService.cs — dùng Resend
public class EmailService(ResendClient resend)
    : IEmailService
{
    public async Task SendChangelogNotificationAsync(
        ChangelogEntry entry,
        IEnumerable<Subscriber> subscribers)
    {
        foreach (var sub in subscribers)
        {
            await resend.Emails.SendAsync(new()
            {
                From = "updates@changelogapp.io",
                To = [sub.Email],
                Subject = $"New update: {entry.Title}",
                Html = BuildEmailHtml(entry, sub)
            });
        }
    }
}

Program.cs — Setup
var builder = WebApplication.CreateBuilder(args);

// Application — MediatR + FluentValidation + ValidationBehavior
builder.Services.AddApplication();

// Infrastructure — DbContext + IAppDbContext, Auth, Email, Stripe, Cache
builder.Services.AddInfrastructure(builder.Configuration);

// Auth — JWT
builder.Services
    .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(o => {
        o.TokenValidationParameters = new() {
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = new SymmetricSecurityKey(
                Encoding.UTF8.GetBytes(
                    builder.Configuration["Jwt:Key"]!))
        };
    });

// Hangfire background jobs
builder.Services.AddHangfire(c => c
    .UsePostgreSqlStorage(builder.Configuration
        .GetConnectionString("DefaultConnection")));
builder.Services.AddHangfireServer();

// Rate limiting
builder.Services.AddRateLimiter(o =>
    o.AddFixedWindowLimiter("widget", w => {
        w.Window = TimeSpan.FromMinutes(1);
        w.PermitLimit = 60;
    }));

var app = builder.Build();
app.UseMiddleware<ExceptionMiddleware>();
app.UseAuthentication();
app.UseAuthorization();

// Map endpoints
app.MapAuthEndpoints();
app.MapEntryEndpoints();
app.MapWidgetEndpoints();
app.MapWebhookEndpoints();

app.Run();

// EntryEndpoints.cs — Minimal API + MediatR ISender
public static class EntryEndpoints
{
    public static void MapEntryEndpoints(
        this WebApplication app)
    {
        var g = app.MapGroup("/api/entries")
            .RequireAuthorization();

        g.MapPost("/", async (
            CreateEntryCommand cmd,
            ISender sender,
            CancellationToken ct) =>
        {
            var id = await sender.Send(cmd, ct);
            return Results.Created($"/api/entries/{id}", id);
        });

        g.MapPost("/{id}/publish", async (
            Guid id,
            ClaimsPrincipal user,
            ISender sender,
            CancellationToken ct) =>
        {
            await sender.Send(
                new PublishEntryCommand(id, user.GetUserId()), ct);
            return Results.Ok();
        });
    }
}

// WidgetEndpoints.cs — public, cached
public static class WidgetEndpoints
{
    public static void MapWidgetEndpoints(
        this WebApplication app)
    {
        app.MapGet("/api/widget/{slug}", async (
            string slug,
            ICacheService cache,
            AppDbContext db) =>
        {
            var cached = await cache
                .GetAsync<WidgetDto>($"widget:{slug}");
            if (cached != null) return cached;

            var data = await db.Projects
                .Where(p => p.Slug == slug)
                .Select(p => new WidgetDto { ... })
                .FirstOrDefaultAsync();

            await cache.SetAsync($"widget:{slug}",
                data, TimeSpan.FromMinutes(5));
            return data;
        }).RequireRateLimiting("widget");
    }
}
