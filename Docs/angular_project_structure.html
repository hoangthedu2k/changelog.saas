auth.service.ts
@Injectable({ providedIn: 'root' })
export class AuthService {
  private router = inject(Router);
  private api = inject(ApiService);

  // Angular Signal — reactive state
  currentUser = signal<User | null>(null);
  isLoggedIn = computed(() =>
    this.currentUser() !== null);

  login(email: string, password: string) {
    return this.api.post<LoginResponse>(
      '/auth/login', { email, password }
    ).pipe(
      tap(res => {
        localStorage.setItem('token', res.token);
        this.currentUser.set(res.user);
      })
    );
  }

  logout() {
    localStorage.removeItem('token');
    this.currentUser.set(null);
    this.router.navigate(['/login']);
  }

  getToken(): string | null {
    return localStorage.getItem('token');
  }

  initFromStorage() {
    const token = this.getToken();
    if (token) {
      // Decode JWT payload để lấy user info
      const payload = JSON.parse(
        atob(token.split('.')[1]));
      this.currentUser.set(payload);
    }
  }
}

// auth.interceptor.ts — tự gắn JWT
export const authInterceptor: HttpInterceptorFn =
  (req, next) => {
    const token = inject(AuthService).getToken();
    if (token) {
      req = req.clone({
        setHeaders: {
          Authorization: `Bearer ${token}`
        }
      });
    }
    return next(req);
  };

// error.interceptor.ts — xử lý 401/403
export const errorInterceptor: HttpInterceptorFn =
  (req, next) => {
    return next(req).pipe(
      catchError(err => {
        if (err.status === 401)
          inject(AuthService).logout();
        if (err.status === 403)
          inject(Router).navigate(['/upgrade']);
        return throwError(() => err);
      })
    );
  };

// api.service.ts — wrapper cho HttpClient
@Injectable({ providedIn: 'root' })
export class ApiService {
  private http = inject(HttpClient);
  private base = environment.apiUrl;

  get<T>(path: string) {
    return this.http.get<T>(
      `${this.base}${path}`);
  }
  post<T>(path: string, body: unknown) {
    return this.http.post<T>(
      `${this.base}${path}`, body);
  }
  put<T>(path: string, body: unknown) {
    return this.http.put<T>(
      `${this.base}${path}`, body);
  }
  delete<T>(path: string) {
    return this.http.delete<T>(
      `${this.base}${path}`);
  }
}

entries.service.ts
@Injectable({ providedIn: 'root' })
export class EntriesService {
  private api = inject(ApiService);

  // Signal-based state
  entries = signal<ChangelogEntry[]>([]);
  loading = signal(false);

  loadEntries(projectId: string) {
    this.loading.set(true);
    return this.api
      .get<ChangelogEntry[]>(
        `/projects/${projectId}/entries`)
      .pipe(
        tap(data => this.entries.set(data)),
        finalize(() => this.loading.set(false))
      );
  }

  createEntry(dto: CreateEntryDto) {
    return this.api
      .post<ChangelogEntry>('/entries', dto)
      .pipe(
        tap(entry => this.entries.update(
          list => [entry, ...list]))
      );
  }

  publishEntry(id: string) {
    return this.api
      .post(`/entries/${id}/publish`, {})
      .pipe(
        tap(() => this.entries.update(list =>
          list.map(e => e.id === id
            ? { ...e, status: 'published' }
            : e)
        ))
      );
  }

  deleteEntry(id: string) {
    return this.api.delete(`/entries/${id}`)
      .pipe(
        tap(() => this.entries.update(
          list => list.filter(e => e.id !== id)))
      );
  }
}

@Component({
  selector: 'app-entry-editor',
  standalone: true,
  imports: [ReactiveFormsModule, CommonModule],
  templateUrl: './entry-editor.component.html'
})
export class EntryEditorComponent {
  private svc = inject(EntriesService);
  private router = inject(Router);
  private route = inject(ActivatedRoute);

  saving = signal(false);
  publishing = signal(false);

  form = new FormGroup({
    title:       new FormControl('',
                   [Validators.required]),
    contentHtml: new FormControl('',
                   [Validators.required]),
    tags:        new FormControl<string[]>([]),
    version:     new FormControl('')
  });

  saveDraft() {
    if (this.form.invalid) return;
    this.saving.set(true);
    this.svc.createEntry(this.form.value as any)
      .subscribe({
        next: (e) => this.router.navigate(
          ['/entries', e.id]),
        error: () => this.saving.set(false)
      });
  }

  publish(entryId: string) {
    this.publishing.set(true);
    this.svc.publishEntry(entryId)
      .subscribe(() => this.publishing.set(false));
  }
}

@Component({
  selector: 'app-changelog-page',
  standalone: true,
  imports: [CommonModule, EntryCardComponent],
  templateUrl: './changelog-page.component.html'
})
export class ChangelogPageComponent
  implements OnInit {

  private route = inject(ActivatedRoute);
  private api = inject(ApiService);
  private meta = inject(Meta);
  private title = inject(Title);

  entries = signal<ChangelogEntry[]>([]);
  project = signal<Project | null>(null);
  selectedTag = signal<string | null>(null);

  filteredEntries = computed(() => {
    const tag = this.selectedTag();
    if (!tag) return this.entries();
    return this.entries()
      .filter(e => e.tags.includes(tag));
  });

  ngOnInit() {
    const slug = this.route.snapshot
      .params['slug'];

    this.api.get<PublicPageDto>(
      `/public/${slug}/entries`)
      .subscribe(data => {
        this.entries.set(data.entries);
        this.project.set(data.project);
        // SEO meta tags
        this.title.setTitle(
          `${data.project.name} Changelog`);
        this.meta.updateTag({
          name: 'description',
          content: `Latest updates for ${data.project.name}`
        });
      });
  }
}

// auth.guard.ts — kiểm tra đăng nhập
export const authGuard: CanActivateFn =
  (route, state) => {
    const auth = inject(AuthService);
    const router = inject(Router);

    if (auth.isLoggedIn()) return true;

    return router.createUrlTree(['/login'],
      { queryParams: { returnUrl: state.url } });
  };

// plan.guard.ts — kiểm tra Pro/Team plan
export const planGuard = (
  required: 'pro' | 'team'
): CanActivateFn => () => {
  const user = inject(AuthService).currentUser();
  if (!user) return false;
  const plans = { free: 0, pro: 1, team: 2 };
  return plans[user.plan] >= plans[required]
    || inject(Router).createUrlTree(['/upgrade']);
};

// app.config.ts — bootstrap config
export const appConfig: ApplicationConfig = {
  providers: [
    provideRouter(routes,
      withComponentInputBinding(),
      withViewTransitions()),

    provideHttpClient(
      withInterceptors([
        authInterceptor,
        errorInterceptor
      ])),

    // SSR support cho public pages
    provideClientHydration(),

    provideAnimationsAsync()
  ]
};

project.service.ts — active project
@Injectable({ providedIn: 'root' })
export class ProjectService {
  private api = inject(ApiService);

  // Global state — project đang chọn
  projects = signal<Project[]>([]);
  activeProject = signal<Project | null>(null);

  // Computed values
  activeProjectId = computed(() =>
    this.activeProject()?.id);

  canAddProject = computed(() => {
    const user = inject(AuthService).currentUser();
    const count = this.projects().length;
    if (user?.plan === 'team') return true;
    if (user?.plan === 'pro') return count < 3;
    return count < 1; // free: 1 project
  });

  loadProjects() {
    return this.api.get<Project[]>('/projects')
      .pipe(
        tap(data => {
          this.projects.set(data);
          if (data.length > 0 &&
              !this.activeProject())
            this.activeProject.set(data[0]);
        })
      );
  }

  setActive(project: Project) {
    this.activeProject.set(project);
  }

  createProject(name: string, slug: string) {
    return this.api
      .post<Project>('/projects', { name, slug })
      .pipe(
        tap(p => this.projects.update(
          list => [...list, p]))
      );
  }
}

@Component({ selector: 'app-billing', standalone: true, ... })
export class BillingComponent {
  private api = inject(ApiService);
  loading = signal(false);

  checkout(priceId: string) {
    this.loading.set(true);
    this.api.post<{ url: string }>(
      '/billing/checkout', { priceId })
      .subscribe({
        next: (res) => {
          // Redirect sang Stripe Checkout page
          window.location.href = res.url;
        },
        error: () => this.loading.set(false)
      });
  }

  openPortal() {
    this.api.get<{ url: string }>('/billing/portal')
      .subscribe(res => window.location.href = res.url);
  }
}
